cmake_minimum_required(VERSION 3.16)
project(magnetic_robot LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ignition-gazebo6 REQUIRED)
find_package(ignition-plugin1 REQUIRED COMPONENTS register)
find_package(fcl REQUIRED)
add_library(MagneticRobot SHARED src/MagneticAdhesion.cc src/SteeringLimitMonitor.cc src/RobotDriveController.cc)
target_link_libraries(MagneticRobot PRIVATE ignition-gazebo6::ignition-gazebo6 ignition-plugin1::register)
target_compile_options(MagneticRobot PRIVATE -Wall -Wextra -Wpedantic)
add_executable(validate_geometry src/ValidateGeometry.cc)
target_link_libraries(validate_geometry PRIVATE fcl)
enable_testing()
add_executable(test_surface src/TestSurface.cc)
target_link_libraries(test_surface PRIVATE ignition-gazebo6::ignition-gazebo6)
add_test(NAME finite_surface_queries COMMAND test_surface)
